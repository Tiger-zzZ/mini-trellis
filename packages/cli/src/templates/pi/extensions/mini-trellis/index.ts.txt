import { existsSync, readFileSync, readdirSync } from "node:fs";
import { execFileSync } from "node:child_process";
import { dirname, join } from "node:path";
import { homedir } from "node:os";

type JsonObject = Record<string, unknown>;

interface PiExtensionContext {
  ui?: {
    notify?: (msg: string, type?: "info" | "warning" | "error") => void;
  };
}

function exists(p: string): boolean {
  try {
    return existsSync(p);
  } catch {
    return false;
  }
}

function findRoot(start: string): string {
  let dir = start;
  for (let i = 0; i < 12; i++) {
    if (exists(join(dir, ".trellis"))) return dir;
    const parent = dirname(dir);
    if (parent === dir) break;
    dir = parent;
  }
  return start;
}

function specIndexes(root: string): string[] {
  const specDir = join(root, ".trellis", "spec");
  if (!exists(specDir)) return [];
  try {
    return readdirSync(specDir).sort().filter((name) =>
      exists(join(specDir, name, "index.md")),
    ).map(name => `.trellis/spec/${name}/index.md`);
  } catch {
    return [];
  }
}

function readDeveloper(root: string): string | null {
  try {
    const p = join(root, ".trellis", ".developer");
    if (!exists(p)) return null;
    const name = readFileSync(p, "utf-8")
      .split(/\r?\n/)
      .find((line) => line.startsWith("name="))
      ?.slice("name=".length)
      .trim();
    return name || null;
  } catch {
    return null;
  }
}

function activeJournal(root: string): string | null {
  const developer = readDeveloper(root);
  if (!developer) return null;
  const workspaceDir = join(root, ".trellis", "workspace", developer);
  if (!exists(workspaceDir)) return null;
  try {
    const journals = readdirSync(workspaceDir)
      .filter((name) => /^journal-\d+\.md$/.test(name))
      .sort(
        (a, b) =>
          Number(a.match(/\d+/)?.[0] || 0) - Number(b.match(/\d+/)?.[0] || 0),
      );
    const journal = journals[journals.length - 1];
    return journal ? `.trellis/workspace/${developer}/${journal}` : null;
  } catch {
    return null;
  }
}

function buildStartupContext(root: string): string {
  const journal = activeJournal(root);
  let entry: string;
  try {
    const script = join(root, ".trellis", "scripts", "common", "knowledge_catalog.py");
    entry = execFileSync(process.platform === "win32" ? "python" : "python3", ["-I", "-X", "utf8", script], {
      cwd: root, input: specIndexes(root).join("\n"), encoding: "utf8", timeout: 3000,
    }).trimEnd();
  } catch {
    entry = "Open .trellis/index.md if present, or the relevant .trellis/spec/ index.";
  }
  return `<session-context>
mini-trellis memory is available on demand. Do not announce this hook.
Before a design, behavior change or project-specific answer, open the relevant entry if you need project knowledge.
For remaining gaps use \`mini-trellis mem recall <query>\`. Expand missing context with \`mini-trellis mem read <ref>\`.
If needed, try one rewrite using a term from the project. Stop when the evidence suffices. On a confirmed or corrected rule, run \`/mini-trellis-remember\` unless asked not to save.
${[journal && `Journal: ${journal}`, entry].filter(Boolean).join("\n")}
</session-context>`;
}

export default function trellisExtension(pi: {
  on?: (
    event: string,
    handler: (event: unknown, ctx?: PiExtensionContext) => unknown,
  ) => void;
}): void {
  if (process.env.TRELLIS_SUBAGENT_CHILD === "1") return;
  const root = findRoot(process.cwd() || homedir());

  pi.on?.("before_agent_start", (event) => {
    const cur = (event as { systemPrompt?: string }).systemPrompt ?? "";
    const startup = buildStartupContext(root);
    const previous = /(?:\n\n)?<session-context>\nmini-trellis memory is available on demand\.[\s\S]*?<\/session-context>/g;
    return {
      systemPrompt: [cur.replace(previous, ""), startup].filter(Boolean).join("\n\n"),
    };
  });

  pi.on?.("context", () => {
    // keep extension loaded; no extra injection
  });

  void (0 as unknown as JsonObject);
}
